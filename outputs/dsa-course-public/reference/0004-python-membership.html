The interview syntax for asking whether a value is already present.

Core Syntax

if value in seen:
    return True

if value not in seen:
    seen.add(value)

Set Pattern

seen = set()

for value in nums:
    if value in seen:
        return True
    seen.add(value)

return False

Dictionary Pattern

index_by_value = {}

for i, value in enumerate(nums):
    if value in index_by_value:
        return index_by_value[value]
    index_by_value[value] = i
