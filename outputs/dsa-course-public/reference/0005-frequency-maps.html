Frequency Maps

Use a dictionary when the count of each value matters.

Manual Counting

counts = {}

for value in values:
    counts[value] = counts.get(value, 0) + 1

Anagram Template

def same_counts(a, b):
    if len(a) != len(b):
        return False

    counts = {}
    for ch in a:
        counts[ch] = counts.get(ch, 0) + 1

    for ch in b:
        if ch not in counts:
            return False
        counts[ch] -= 1
        if counts[ch] == 0:
            del counts[ch]

    return len(counts) == 0
